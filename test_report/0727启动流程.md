
预期为：

```text

```

控制链路静止基线：

```bash
python3 - <<'PY'
import math, time
import rclpy
from rclpy.node import Node
from geometry_msgs.msg import Twist, Vector3

TOPICS = {
    '/cmd_vel_nav': Twist,
    '/cmd_vel_smoothed': Twist,
    '/cmd_vel': Twist,
    '/mx_base_vel_command': Vector3,
}

class Watch(Node):
    def __init__(self):
        super().__init__('internnav_quiet_probe')
        self.stats = {t: {'samples': 0, 'nonzero': 0} for t in TOPICS}
        for topic, typ in TOPICS.items():
            self.create_subscription(typ, topic, lambda msg, topic=topic: self.cb(topic, msg), 10)

    def cb(self, topic, msg):
        if hasattr(msg, 'linear'):
            lx, ly, az = float(msg.linear.x), float(msg.linear.y), float(msg.angular.z)
        else:
            lx, ly, az = float(msg.x), float(msg.y), float(msg.z)
        self.stats[topic]['samples'] += 1
        if math.hypot(lx, ly) > 1e-4 or abs(az) > 1e-4:
            self.stats[topic]['nonzero'] += 1

rclpy.init()
node = Watch()
end = time.time() + 3.0
while time.time() < end:
    rclpy.spin_once(node, timeout_sec=0.1)
for topic, stat in node.stats.items():
    print(topic, 'samples', stat['samples'], 'nonzero', stat['nonzero'])
node.destroy_node()
rclpy.shutdown()
PY
```

验收：四个控制话题 `nonzero=0`。
